 1.Palindrome Number 

import java.util.Scanner;

public class palindromenumber {
	static class palindromeThread extends Thread {
		String s;

		palindromeThread(String s) {
			this.s = s;
		}

		public void run() {
			try {
				String rev = "";
				for (int i = s.length() - 1; i >= 0; i--) {
					rev = rev + s.charAt(i);
				}
				if (s.equals(rev)) {
					System.out.println("it is Palindrome Number");
				} else {
					System.out.println("it is not a Palindrome Number ");
				}
			} catch (Exception e) {
				System.out.println("Error ..." + e);
			}
		}
	}

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		try {
			System.out.println("Enter the Number :  ");
			String input = sc.next();
			palindromeThread t = new palindromeThread(input);
			t.start();
		} catch (Exception e) {
			System.out.println("Main Error :" + e);
		} finally {
			sc.close();
		}

	}
}

		}

		public void run() {
			try {
				boolean isprime = true;
				if (n <= 1) {
					isprime = false;

3.Armstrong Number 



import java.util.Scanner;

public class Armstrongnumber {
	static class ArmstrongThread extends Thread {
		int n;

		ArmstrongThread(int n) {
			this.n = n;
		}

		public void run() {
			try {
				int sum = 0;
				int temp = n;
				while (temp > 0) {
					int digit = temp % 10;
					sum += (digit * digit * digit);
					temp /= 10;
				}
				if (sum == n) {
					System.out.println("it is Amstrong Number ");
				} else {
					System.out.println("it is not a Amstrong number ");

	static class fibonacciThread extends Thread {
		int n;

		fibonacciThread(int n) {
			this.n = n;
		}

		public void run() {
			try {
				int first = 0;
				int secound = 1;
				for (int i = 0; i <= n; i++) {
					System.out.println(first + "");

					secound = temp;
				}

import java.util.Scanner;

public class ReversingNumber {
	static class reversThread extends Thread {
		int n;

		reversThread(int n) {
			this.n = n;

		}

		public void run() {
			try {

				int temp = n;
				int reversed = 0;
				while (temp != 0) {
					int digit = temp % 10;
					reversed = reversed * 10 + digit;
					temp /= 10;
				}
				System.out.println("Reversed Number : " + reversed);
			} catch (Exception e) {

7.Swapping Number



import java.util.Scanner;

public class SwappingNumber {
	static class Data {
		int a, b;

		Data(int a, int b) {
			this.a = a;
			this.b = b;
		}
	}

	static class swappingThread extends Thread {
		Data data;

		swappingThread(Data data) {
			this.data = data;
		}

		public void run() {
			try {
				synchronized (data) {
					data.a = data.a + data.b;
					data.b = data.a - data.b;
					data.a = data.a - data.b;

					System.out.println("A : " + data.a);
					System.out.println("B : " + data.b);
				}
			} catch (Exception e) {
				System.out.println("Error..." + e);
			}
		}
	}

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		try {
			System.out.println("Enter the number A: ");
			int a = sc.nextInt();
			System.out.println("Enter the number  B: ");
			int b = sc.nextInt();
			Data data = new Data(a, b);
			swappingThread t = new swappingThread(data);
			t.start();
		} catch (Exception e) {
			System.out.println("Main Error ...." + e);
		} finally {
			sc.close();
		}
	}
}
