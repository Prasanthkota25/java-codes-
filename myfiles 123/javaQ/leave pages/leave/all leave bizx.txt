document.getElementById("leaveEntryForm").onsubmit = function (e) {

    e.preventDefault();

    if (!currentUser) return alert("User not loaded");

    const leaveType = document.getElementById("leaveType").value;
    const fromDate = document.getElementById("startDate").value;
    const toDate = document.getElementById("endDate").value;
    const session = document.getElementById("session").value;
    const noOfDays = parseFloat(document.getElementById("noOfDays").value);
    const reason = document.getElementById("reasonArea").value.trim();

    if (!leaveType || !fromDate || !toDate || !reason)
        return alert("Fill all fields");

    if (leaveType === "Privilege" && noOfDays > balances.privilege)
        return alert("No Privilege balance");

    if (leaveType === "Sick" && noOfDays > balances.sick)
        return alert("No Sick balance");

    if (leaveType === "Comp-Off Leave" && noOfDays > balances.compOff)
        return alert("No CompOff balance");

    // deduct balance
    if (leaveType === "Privilege") balances.privilege -= noOfDays;
    if (leaveType === "Sick") balances.sick -= noOfDays;
    if (leaveType === "Comp-Off Leave") balances.compOff -= noOfDays;

    saveBalances();

    const leave = {
        employeeName: document.getElementById("fullName").value,
        username: currentUser.username,
        leaveType,
        fromDate,
        toDate,
        sessionDisplay: session,
        noOfDays,
        reason,
        status: "Pending"
    };

    const history = loadHistory();
    history.unshift(leave);

    saveHistory(history);

    renderHistory();

    alert("Leave Applied Successfully!");
    resetForm();
};

/* ======================
   RENDER HISTORY (USER ONLY)
====================== */
function renderHistory() {

    const table = document.getElementById("historyBody");
    table.innerHTML = "";

    const data = loadHistory();

    data.forEach((item, i) => {

        const row = document.createElement("tr");

        row.innerHTML = `
            <td>${item.employeeName}</td>
            <td>${item.leaveType}</td>
            <td>${item.fromDate}</td>
            <td>${item.toDate}</td>
            <td>${item.sessionDisplay}</td>
            <td>${item.noOfDays}</td>
            <td>${item.reason}</td>
            <td>${item.status}</td>
            <td>
                <i class="fa-solid fa-trash" onclick="deleteRow(${i})"></i>
            </td>
        `;

        table.appendChild(row);
    });
}

/* ======================
   DELETE (FIXED PER USER)
====================== */
function deleteRow(index) {

    let data = loadHistory();
    const item = data[index];

    if (item.leaveType === "Privilege") balances.privilege += item.noOfDays;
    if (item.leaveType === "Sick") balances.sick += item.noOfDays;
    if (item.leaveType === "Comp-Off Leave") balances.compOff += item.noOfDays;

    saveBalances();

    data.splice(index, 1);
    saveHistory(data);

    renderHistory();
    updateLeaveBalance();

    alert("Deleted Successfully");
}
