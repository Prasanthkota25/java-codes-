public class all {

	public static void main(String[] args) {
		String s = "prasanth";
		char arr[] = s.toCharArray();
		int left = 0, right = s.length() - 1;
		while (left < right) {
			char temp = arr[left];
			arr[left] = arr[right];
			arr[right] = temp;
			left++;
			right--;

		}
		System.out.println(new String(arr));

public class all {

	public static void main(String[] args) {
		String s = "java";
		s = s.toLowerCase();

		int vowels = 0;
		int consonents = 0;
		for (int i = 0; i < s.length(); i++) {
			char ch = s.charAt(i);
			if (ch >= 'a' && ch <= 'z') {
				if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u') {
					vowels++;
				} else {
					consonents++;
				}
			}
		}
		System.out.println("vowesl : " + vowels);
		System.out.println("consonents :" + consonents);
	}
}
