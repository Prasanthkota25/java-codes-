	public static int lengthOfSubString(String s) {
		int[] last = new int[128];
		for (int i = 0; i<128; i++) {
			last[i] = -1;
		}
		
		int start = 0;
		int Max = 0;
		
		for (int i = 0; i<s.length(); i++) {
			char c = s.charAt(i);
			
			start = Math.max(start, last[c] + 1);
			last[c] = i;
			Max = Math.max(Max, i - start + 1);
		}

		return Max;
	}

}
